const BASE_URL = import.meta.env.VITE_API_URL || "http://localhost:5000/api";
const API_ORIGIN = BASE_URL.replace(/\/api\/?$/, "");
const TOKEN_KEY = "elearn.token";

export const getToken = () => localStorage.getItem(TOKEN_KEY);
export const setToken = (t: string) => localStorage.setItem(TOKEN_KEY, t);
export const clearToken = () => localStorage.removeItem(TOKEN_KEY);

export const absoluteUploadUrl = (p?: string | null) => {
  if (!p) return "";
  if (p.startsWith("http")) return p;
  return `${API_ORIGIN}${p.startsWith("/") ? "" : "/"}${p}`;
};

type ApiEnvelope<T> = { success: boolean; statusCode: number; message: string; data: T };

async function request<T>(path: string, options: RequestInit = {}): Promise<T> {
  const token = getToken();
  const headers: Record<string, string> = { ...(options.headers as Record<string, string>) };
  if (!(options.body instanceof FormData)) headers["Content-Type"] = "application/json";
  if (token) headers.Authorization = `Bearer ${token}`;

  const res = await fetch(`${BASE_URL}${path}`, { ...options, headers });
  const json = (await res.json().catch(() => ({}))) as Partial<ApiEnvelope<T>>;
  if (!res.ok) throw new Error(json.message || `Request failed (${res.status})`);
  return (json.data as T) ?? ({} as T);
}

export const api = {
  signup: (body: { name: string; email: string; password: string; phone?: string }) =>
    request<{ user: BackendUser; token: string }>("/auth/signup", { method: "POST", body: JSON.stringify(body) }),
  login: (body: { email: string; password: string }) =>
    request<{ user: BackendUser; token: string }>("/auth/login", { method: "POST", body: JSON.stringify(body) }),
  me: () => request<{ user: BackendUser }>("/auth/me"),

  listUsers: (params: { page?: number; limit?: number; search?: string; role?: "user" | "admin" } = {}) => {
    const q = new URLSearchParams();
    if (params.page) q.set("page", String(params.page));
    if (params.limit) q.set("limit", String(params.limit));
    if (params.search) q.set("search", params.search);
    if (params.role) q.set("role", params.role);
    return request<{ users: BackendUserRow[]; total: number; page: number; limit: number }>(
      `/users${q.toString() ? "?" + q.toString() : ""}`
    );
  },
  updateUserStatus: (id: number, isActive: boolean) =>
    request<{ user: BackendUserRow }>(`/users/${id}/status`, { method: "PATCH", body: JSON.stringify({ isActive }) }),

  listCategories: (includeInactive = false) =>
    request<{ categories: BackendCategory[] }>(`/categories?includeInactive=${includeInactive}`),
  createCategory: (body: { name: string; shortHeading?: string; description: string; sortOrder?: number }) =>
    request<{ category: BackendCategory }>("/categories", { method: "POST", body: JSON.stringify(body) }),
  updateCategory: (id: number, patch: Partial<{ name: string; shortHeading: string; description: string; sortOrder: number; isActive: boolean }>) =>
    request<{ category: BackendCategory }>(`/categories/${id}`, { method: "PATCH", body: JSON.stringify(patch) }),
  deleteCategory: (id: number) => request<{}>(`/categories/${id}`, { method: "DELETE" }),

  listCourses: (params: { includeAll?: boolean; categoryId?: number; search?: string } = {}) => {
    const q = new URLSearchParams();
    if (params.includeAll) q.set("includeAll", "true");
    if (params.categoryId) q.set("categoryId", String(params.categoryId));
    if (params.search) q.set("search", params.search);

      </div>
    </div>
  );
}

function Router() {
  const { route } = useShop();
  const base = route.split("?")[0];

  useEffect(() => {
    const key = base.split("/")[0] || "home";
    document.title = titles[key] ?? "Elearn Hub";
  }, [base]);

  const renderPage = () => {
    if (base === "" || base === "home") return <Home />;
    if (base === "courses") return <Courses />;
    if (base.startsWith("course/")) {
      const id = Number(base.split("/")[1] ?? 0);
      return <CourseDetail id={id} />;
    }
    if (base === "paths") return <Paths />;
    if (base === "mentor") return <Mentor />;
    if (base === "reviews") return <Reviews />;
    if (base === "about") return <About />;
    if (base === "contact" || base === "teams") return <Contact />;
    if (base === "cart") return <CartPage />;
    if (base === "wishlist") return <Wishlist />;
    if (base === "dashboard") return <Dashboard />;
    if (base === "admin") return <Admin />;
    if (base.startsWith("legal/")) {
      const doc = base.split("/")[1] ?? "terms";
      return <Legal doc={doc} />;
    }
    return <NotFound />;
  };

  return (
    <div className="flex min-h-screen flex-col bg-cream">
      <a
        href="#main-content"
        className="sr-only focus:not-sr-only focus:absolute focus:left-4 focus:top-4 focus:z-[100] focus:rounded-full focus:bg-ink focus:px-4 focus:py-2 focus:text-sm focus:text-white"
      >
        Skip to content
      </a>
      <Header />
      <main id="main-content" className="flex-1">
        {renderPage()}
      </main>
      {base !== "dashboard" && base !== "admin" && <Footer />}
      <CartDrawer />
      <AuthModal />
      <Toasts />
    </div>
  );
}

export default function App() {
  return (
    <ShopProvider>
      <Router />
    </ShopProvider>
  );
}

/* ---------------- ProgressRing ---------------- */
export function ProgressRing({
  value,
  size = 84,
  stroke = 8,
  tint = "#CDF56A",
  label,
}: {
  value: number;
  size?: number;
  stroke?: number;
  tint?: string;
  label?: ReactNode;
}) {
  const r = (size - stroke) / 2;
  const circ = 2 * Math.PI * r;
  const clamped = Math.max(0, Math.min(100, value));
  const offset = circ - (clamped / 100) * circ;
  return (
    <div className="relative grid place-items-center" style={{ width: size, height: size }}>
      <svg width={size} height={size} className="-rotate-90">
        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="#EFEAE0" strokeWidth={stroke} />
        <circle
          cx={size / 2}
          cy={size / 2}
          r={r}
          fill="none"
          stroke={tint}
          strokeWidth={stroke}
          strokeLinecap="round"
          strokeDasharray={circ}
          strokeDashoffset={offset}
          style={{ transition: "stroke-dashoffset 600ms ease" }}
        />
      </svg>
      {label && <div className="absolute inset-0 grid place-items-center">{label}</div>}
    </div>
  );
}

/* ---------------- Reveal (subtle on-scroll) ---------------- */
export function Reveal({ children, delay = 0 }: { children: ReactNode; delay?: number }) {
  const ref = useRef<HTMLDivElement>(null);
  const [shown, setShown] = useState(false);
  useEffect(() => {
    const el = ref.current;
    if (!el) return;
    const io = new IntersectionObserver(
      ([entry]) => { if (entry.isIntersecting) { setShown(true); io.disconnect(); } },
      { threshold: 0.1 }
    );
    io.observe(el);
    return () => io.disconnect();
  }, []);
  return (
    <div
      ref={ref}
      style={{ transitionDelay: `${delay}ms` }}
      className={cn("transition-all duration-500", shown ? "translate-y-0 opacity-100" : "translate-y-3 opacity-0")}
    >
      {children}
    </div>
  );
}

/* ---------------- useCountdown ---------------- */
export function useCountdown(hoursFromNow: number) {
  const target = useRef(Date.now() + hoursFromNow * 3600 * 1000);
  const [now, setNow] = useState(Date.now());
  useEffect(() => {
    const t = setInterval(() => setNow(Date.now()), 1000);
    return () => clearInterval(t);
  }, []);
  const ms = Math.max(0, target.current - now);
  const totalSec = Math.floor(ms / 1000);
  return {
    dd: Math.floor(totalSec / 86400),
    hh: Math.floor((totalSec % 86400) / 3600),
    mm: Math.floor((totalSec % 3600) / 60),
    ss: totalSec % 60,
  };
}
